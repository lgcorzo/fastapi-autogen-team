41pub async fn route_query(
42    State(state): State<Arc<AppState>>,
43    headers: HeaderMap,
44    ValidatedJson(request): ValidatedJson<Input>,
45) -> impl IntoResponse {
46    // Basic Header Sanitization (example: carry over authorization if needed)
47    let _auth = headers.get("authorization");
48
49    // Validation: Empty messages
50    if request.messages.is_empty() {
51        return (
52            StatusCode::UNPROCESSABLE_ENTITY,
53            Json(json!({
54                "error": "Unprocessable Entity",
55                "details": "messages list cannot be empty"
56            })),
57        )
58            .into_response();
59    }
60
61    if request.stream.unwrap_or(false) {
62        let stream = state.team.run_stream(request).await.unwrap();
63        let sse_stream = stream.map(|res| {
64            let content = res.unwrap_or_else(|e| format!("Error: {}", e));
65            let chunk = json!({
66                "choices": [{
67                    "delta": { "content": content },
68                    "index": 0,
69                    "finish_reason": if content.contains("TERMINATE") { Some("stop") } else { None }
70                }]
71            });
72            Ok::<Event, Infallible>(Event::default().data(chunk.to_string()))
73        });
74        return Sse::new(sse_stream)
75            .keep_alive(axum::response::sse::KeepAlive::default())
76            .into_response();
77    }
78
79    let response = state
80        .team
81        .run(request)
82        .await
83        .unwrap_or_else(|e| format!("Error: {}", e));
84
85    let output = json!({
86        "id": "chatcmpl-default",
87        "object": "chat.completion",
88        "created": chrono::Utc::now().timestamp(),
89        "model": "minimax-m2.7:cloud",
90        "choices": [{
91            "index": 0,
92            "message": {
93                "role": "assistant",
94                "content": response,
95            },
96            "finish_reason": "stop"
97        }],
98        "usage": {
99            "prompt_tokens": 0,
100            "completion_tokens": 0,
101            "total_tokens": 0
102        }
103    });
104
105    Json(output).into_response()
106}
