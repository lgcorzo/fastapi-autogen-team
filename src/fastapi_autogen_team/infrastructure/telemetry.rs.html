1use opentelemetry::KeyValue;
2use opentelemetry_otlp::WithExportConfig;
3use opentelemetry_sdk::{runtime, trace::Config, Resource};
4use tracing_subscriber::{layer::SubscriberExt, util::SubscriberInitExt, EnvFilter};
5
6pub fn init_telemetry(app_name: &str, endpoint: &str) -> anyhow::Result<()> {
7    let resource = Resource::new(vec![KeyValue::new("service.name", app_name.to_string())]);
8
9    let tracer = opentelemetry_otlp::new_pipeline()
10        .tracing()
11        .with_exporter(
12            opentelemetry_otlp::new_exporter()
13                .http()
14                .with_endpoint(format!("{}/traces", endpoint)),
15        )
16        .with_trace_config(Config::default().with_resource(resource))
17        .install_batch(runtime::Tokio)?;
18
19    let telemetry = tracing_opentelemetry::layer().with_tracer(tracer);
20
21    tracing_subscriber::registry()
22        .with(EnvFilter::from_default_env().add_directive(tracing::Level::INFO.into()))
23        .with(telemetry)
24        .with(tracing_subscriber::fmt::layer())
25        .init();
26
27    Ok(())
28}
