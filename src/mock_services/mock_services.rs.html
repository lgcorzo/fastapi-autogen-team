1use axum::{
2    extract::Query,
3    routing::{get, post},
4    Json, Router,
5};
6use serde::Deserialize;
7use serde_json::{json, Value};
8use std::net::SocketAddr;
9use std::sync::Arc;
10use tracing_subscriber::{layer::SubscriberExt, util::SubscriberInitExt};
11
12#[derive(Clone)]
13struct AppState {}
14
15#[tokio::main]
16async fn main() {
17    tracing_subscriber::registry()
18        .with(tracing_subscriber::fmt::layer())
19        .with(tracing_subscriber::EnvFilter::new("info"))
20        .init();
21
22    let state = Arc::new(AppState {});
23
24    let app = Router::new()
25        .route("/", get(|| async { "Mock Services Running" }))
26        // R2R Mocks
27        .route("/v3/users/login", post(r2r_login))
28        .route("/v3/retrieval/rag", post(r2r_rag))
29        .route("/v3/retrieval/search", post(r2r_search))
30        // Jira Mocks
31        .route("/rest/api/2/search", get(jira_search))
32        .with_state(state);
33
34    let addr = SocketAddr::from(([127, 0, 0, 1], 8081));
35    tracing::info!("Mock server listening on {}", addr);
36    let listener = tokio::net::TcpListener::bind(addr).await.unwrap();
37    axum::serve(listener, app).await.unwrap();
38}
