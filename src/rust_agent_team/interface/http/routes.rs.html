1use crate::domain::agent::team::AgentTeam;
2use crate::interface::http::handlers::{docs_redirect, get_models, route_query};
3use crate::interface::http::middleware::{cors_layer, security_headers};
4use axum::{
5    routing::{get, post},
6    Router,
7};
8use std::sync::Arc;
9use tower_http::trace::TraceLayer;
10
11pub struct AppState {
12    pub team: AgentTeam,
13}
14
15pub fn create_app(state: Arc<AppState>) -> Router {
16    let mut router = Router::new()
17        .route("/agent", get(docs_redirect))
18        .nest(
19            "/agent/api/v1beta",
20            Router::new()
21                .route("/models", get(get_models))
22                .route("/chat/completions", post(route_query)),
23        )
24        .with_state(state);
25
26    // Apply security headers
27    for layer in security_headers() {
28        router = router.layer(layer);
29    }
30
31    // Apply CORS
32    if let Some(cors) = cors_layer() {
33        router = router.layer(cors);
34    }
35    router.layer(TraceLayer::new_for_http())
36}
