3pub async fn get_confluence_results(url: &str, query: &str) -> anyhow::Result<String> {
4    let user = env::var("JIRA_USERNAME")?;
5    let token = env::var("JIRA_API_TOKEN")?;
6
7    let client = reqwest::Client::new();
8
9    // CQL query
10    let sanitized_query = query.replace('\\', "\\\\").replace('"', "\\\"");
11    let cql = format!(
12        "text ~ \"{}\" OR title ~ \"{}\"",
13        sanitized_query, sanitized_query
14    );
15
16    let base_url = url.trim_end_matches('/');
17    let search_url = if base_url.ends_with("/wiki") {
18        format!("{}/rest/api/content/search", base_url)
19    } else {
20        format!("{}/wiki/rest/api/content/search", base_url)
21    };
22
23    let res = client
24        .get(&search_url)
25        .basic_auth(&user, Some(&token))
26        .query(&[("cql", &cql)])
27        .send()
28        .await?;
29
30    let data: serde_json::Value = res.json().await?;
31    let results = data["results"].as_array();
32
33    match results {
34        Some(list) if !list.is_empty() => {
35            let res_list: Vec<String> = list
36                .iter()
37                .map(|item| {
38                    let title = item["title"].as_str().unwrap_or("No title");
39                    let webui = item["_links"]["webui"].as_str().unwrap_or("");
40                    let full_url = format!("{}{}", base_url, webui);
41                    format!("- {}: {}", title, full_url)
42                })
43                .collect();
44            Ok(res_list.join("\n"))
45        }
46        _ => Ok("No se encontraron resultados en Confluence.".to_string()),
47    }
48}
